    pub async fn resolve_report(
        pool: &PgPool,
        report_id: Uuid,
        new_status: &str,
        resolution_photo_path: &str,
        resolution_notes: Option<&str>,
        changed_by: Uuid,
    ) -> Result<bool, AppError> {
        let mut tx = pool.begin().await?;
        let result = sqlx::query(
            "UPDATE reports SET status = $1::report_status,
             resolution_photo_path = $2, resolution_notes = $3
             WHERE id = $4"
        )
        .bind(new_status)
        .bind(resolution_photo_path)
        .bind(resolution_notes)
        .bind(report_id)
        .execute(&mut *tx)
        .await?;

        if result.rows_affected() == 0 {
            tx.rollback().await?;
            return Ok(false);
        }

        sqlx::query(
            "INSERT INTO status_history (report_id, new_status, note, changed_by)
             VALUES ($1, $2::report_status, $3, $4)"
        )
        .bind(report_id)
        .bind(new_status)
        .bind(resolution_notes)
        .bind(changed_by)
        .execute(&mut *tx)
        .await?;

        tx.commit().await?;
        Ok(true)
    }
