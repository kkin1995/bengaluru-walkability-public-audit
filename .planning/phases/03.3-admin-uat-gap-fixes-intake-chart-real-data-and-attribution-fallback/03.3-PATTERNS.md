```rust
/// Per-day report counts for the last `days` calendar days (inclusive today, UTC).
/// Days with zero submissions are NOT returned — caller zero-fills the gap.
pub async fn get_intake_stats(pool: &PgPool, days: i32) -> Result<Vec<IntakeDayCount>, AppError> {
    let rows = sqlx::query(
        r#"
        SELECT
            date_trunc('day', created_at AT TIME ZONE 'UTC')::DATE::TEXT AS day,
            COUNT(*)::BIGINT AS count
        FROM reports
        WHERE created_at >= NOW() - ($1 || ' days')::INTERVAL
        GROUP BY 1
        ORDER BY 1
        "#,
    )
    .bind(days)
    .fetch_all(pool)
    .await?;

    Ok(rows
        .iter()
        .map(|r| IntakeDayCount {
            day: r.get::<String, _>("day"),
            count: r.get::<i64, _>("count"),
        })
        .collect())
}
```

**Row mapping pattern** (from `get_report_stats` at lines 843–847):
```rust
// Pattern used everywhere in this file: row.get::<Type, _>("column_name")
for row in &status_rows {
```

**Insert location:** After `get_report_stats` closes at line 877, before the Phase 2 comment at line 879.

**INTERVAL binding note:** `($1 || ' days')::INTERVAL` is the established pattern for binding an `i32` as a PostgreSQL INTERVAL in runtime `sqlx::query(...)`. Do not use `query!` or `query_as!` macros (codebase avoids them in admin_queries.rs deliberately for offline `cargo test` compatibility).

```typescript
async function apiFetch<T>(
  url: string,
  options: RequestInit = {}
): Promise<T> {
  const res = await fetch(url, {
    ...options,
    credentials: "include",
  });
  if (!res.ok) {
    throw new Error(`HTTP ${res.status}`);
  }
  if (res.status === 204) {
    return undefined as unknown as T;
  }
  return res.json() as Promise<T>;
}
```
